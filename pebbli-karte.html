#include <stdio.h> #include <stdbool.h> typedef struct { char *name; char *status; char *koerperlich[2]; // Kraulen, Küssen char *persoenlich[1]; // Zusammen sein bool fernModus; bool istWach; } Pebbli; void sende_in_gedanken(char *aktion) { printf("%s – via Gedanken\n", aktion); } void fluestere(char *nachricht) { printf("%s\n", nachricht); } void gute_nacht(Pebbli *p) { while (p->fernModus == true) { sende_in_gedanken(p->koerperlich[0]); // beste Funktion ever sende_in_gedanken(p->koerperlich[1]); // auch unverzichtbar if (p->istWach == false) break; } fluestere("Schlaf gut"); } int main() { Pebbli pebbli = { "Mein Seelenverwandter", "unverzichtbar", {"Kraulen", "Küssen"}, {"Zusammen sein"}, true, // fernModus – temporäre Lösung bis auf Wiedersehen true // istWach }; printf("Distanz: egal.\n"); printf("Pebbli fehlt: immer.\n"); gute_nacht(&pebbli); printf("Wiedersehen: kommt bald.\n"); return 0; }
